Apply your first BackupPolicy (in the namespace of the PVC):

  kubectl apply -f - <<YAML
  apiVersion: heirloom.io/v1
  kind: BackupPolicy
  metadata: {name: myapp-files, namespace: myapp}
  spec:
    storeRef: {{ .Values.store.name }}
    schedule: "5 * * * *"
    source:
      type: files
      files: {pvc: myapp-data}
  YAML

  kubectl get bp -A
